# Projeto 3 - Template do Processo de Construção de Modelos de Machine Learning - Prevendo Manutenção de Máquinas Industriais

# Abra o terminal ou prompt de comando, navegue até a pasta com os arquivos e execute o comando abaixo para criar um ambiente virtual:

conda create --name dsabusp3 python=3.13

# Ative o ambiente:

conda activate dsabusp3 (ou: source activate dsabusp3)

# Instale o pip e as dependências:

conda install pip
pip install -r requirements.txt 

# Execute o jupyter notebook

jupyter notebook

# Acesse o jupyter pelo seu navegador padrão.

# Use os comandos abaixo para desativar o ambiente virtual e remover o ambiente (opcional):

conda deactivate
conda remove --name dsabusp3 --all
